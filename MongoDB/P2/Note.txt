    SCHEMA VALIDATIONS: 

    SchemaType options

    const bookSchema = mongoose.schema({
        tittle:{
            type:String,
            required:true,
        },
        author:{
            type:String,
        },
        price:{
            type:Number,
            min:[1,"Please enter valid price"] ,//min:1
        },
        discount:{
            type:Number,
            default:0,
        },
        genre:[String],
        category:{
            type:String,
            enum:["fiction","non-fiction"],
        },
    });

    let Bookx = mongoose.model({
        name:"Eat That Frog",
        author:"Brian Tracy",
        price:520,
        genre["self improvement"],
        category:"non-fiction"
    })

    Bookx.save().then(res=>{
        console.log(res);
    }).catch(e=>{
        console.log(e);
    })
