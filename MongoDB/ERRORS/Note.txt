    custom:
    class ExpressError exteds Error{
        constructor(status,message){
            super();
            this.status = status;
            this.message = message;
        }
    }

    app.use((err,req,res,next)=>{
        let {status = 500,message="SOME ERROR"} = err;
        res.status(status).send(message);
    });
